**描述**：

- 给你一个整数数组 `nums`，请你将该数组升序排列。

**示例 1：**
输入：nums = [5,2,3,1]
输出：[1,2,3,5]
解释：数组升序排列后得到 [1,2,3,5]。

**示例 2：**
输入：nums = [5,1,1,2,0,0]
输出：[0,0,1,1,2,5]
解释：数组升序排列后得到 [0,0,1,1,2,5]。

**提示：**
- 1 <= nums.length <= 5 * 10^4
- -5 * 10^4 <= nums[i] <= 5 * 10^4


**解法**
```java
public int[] sortArray(int[] nums) {
	// quickSort(nums, 0, nums.length - 1);
	tmp = new int[nums.length];
	mergeSort(nums, 0, nums.length - 1);
	return nums;
}


/*
 * 归并排序
 * 1. 将数组递归拆分为左右两半，直到区间长度为 1。
 * 2. 回溯时用临时数组合并两个有序区间，使当前区间整体有序。
 * 3. 通过分治保证每一层处理 n 个元素，共 log n 层。
 */

/*
 * 时间复杂度：O(n logn)
 * 说明：每层合并会遍历当前层全部元素，共有 O(log n) 层。
 *
 * 空间复杂度：O(n)
 * 说明：需要长度为 n 的临时数组 tmp；递归栈为 O(log n)，整体额外空间为 O(n)。
 */
int[] tmp;

void mergeSort(int[] nums, int l, int r) {
	if (l >= r)
		return;
	int mid = (l + r) / 2;
	mergeSort(nums, l, mid);
	mergeSort(nums, mid + 1, r);
	merge(nums, l, mid, r);
}

void merge(int[] nums, int l, int mid, int r) {
	int i = l, j = mid + 1, k = 0;
	while (i <= mid && j <= r) {
		if (nums[i] < nums[j])
			tmp[k++] = nums[i++];
		else
			tmp[k++] = nums[j++];
	}
	while (i <= mid)
		tmp[k++] = nums[i++];
	while (j <= r)
		tmp[k++] = nums[j++];
	for (i = l, k = 0; i <= r;) {
		nums[i++] = tmp[k++];
	}
}


/*
 * 随机化快速排序
 * 1. 随机选择基准值，降低遇到极端划分的概率。
 * 2. partition 将小于基准的元素放左侧，大于基准的元素放右侧。
 * 3. 递归排序基准左右两段区间。
 *
 * 时间复杂度：平均 O(n log n)，最坏 O(n^2)
 * 说明：平均每次划分较均衡；若连续极端划分则退化为平方级。
 *
 * 空间复杂度：平均 O(log n)，最坏 O(n)
 * 说明：主要来自递归调用栈，递归深度取决于划分是否均衡。
 */
Random rand = new Random();
 
void quickSort(int[] nums, int l, int r) {
	if (l >= r)
		return;
	int pivot = partition(nums, l, r);
	quickSort(nums, l, pivot - 1);
	quickSort(nums, pivot + 1, r);
}

int partition(int[] nums, int l, int r) {
	int idx = rand.nextInt(r - l + 1) + l;
	swap(nums, l, idx);
	int pivot = nums[l], start = l;
	l++;
	while (l <= r) {
		while (l <= r && nums[l] < pivot)
			l++;
		while (l <= r && nums[r] > pivot)
			r--;
		if (l >= r)
			break;
		swap(nums, l, r);
		l++;
		r--;
	}
	swap(nums, start, r);
	return r;
}

void swap(int[] nums, int l, int r) {
	int tmp = nums[l];
	nums[l] = nums[r];
	nums[r] = tmp;
}
```
